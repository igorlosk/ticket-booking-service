DTO: Request vs Response

// Входящие данные (клиент → сервер)
public record CreateEventRequest(
    @NotBlank String name,
    @NotNull @Future LocalDateTime date,
    @NotBlank String venue
) {}

// Исходящие данные (сервер → клиент)
public record EventResponse(
    Long id,
    String name,
    LocalDateTime date,
    String venue,
    Status status
) {}

Почему разделять:

  • В CreateEventRequest нет id — его генерит сервер
  • В EventResponse нет валидации — она нужна только на входе
  • Клиент не может подсунуть свой id при создании

Шаг 3. Repository (in-memory)

@Repository
public class EventRepository {
    private final Map<Long, Event> storage = new ConcurrentHashMap<>();
    private final AtomicLong idGenerator = new AtomicLong(0);

    public Event save(Event event) {
        if (event.getId() == null) event.setId(idGenerator.incrementAndGet());
        storage.put(event.getId(), event);
        return event;
    }
    public Optional<Event> findById(Long id) { return Optional.ofNullable(storage.get(id)); }
    public List<Event> findAll() { return new ArrayList<>(storage.values()); }
    public void deleteById(Long id) { storage.remove(id); }
    public boolean existsById(Long id) { return storage.containsKey(id); }
}

Шаг 4. Service

@Service
public class EventService {
    private final EventRepository eventRepository;

    public EventService(EventRepository eventRepository) {
        this.eventRepository = eventRepository;
    }

    public EventResponse create(CreateEventRequest req) {
        Event event = new Event(null, req.name(), req.date(), req.venue(), Status.AVAILABLE);
        return EventResponse.from(eventRepository.save(event));
    }

    public EventResponse findById(Long id) {
        Event event = eventRepository.findById(id)
                .orElseThrow(() -> new EventNotFoundException(id));
        return EventResponse.from(event);
    }

    public List<EventResponse> findAll() {
        return eventRepository.findAll().stream().map(EventResponse::from).toList();
    }

    public EventResponse update(Long id, UpdateEventRequest req) {
        Event event = eventRepository.findById(id)
                .orElseThrow(() -> new EventNotFoundException(id));
        event.setName(req.name());
        event.setDate(req.date());
        event.setVenue(req.venue());
        event.setStatus(req.status());
        return EventResponse.from(eventRepository.save(event));
    }

    public void delete(Long id) {
        if (!eventRepository.existsById(id)) throw new EventNotFoundException(id);
        eventRepository.deleteById(id);
    }
}

Шаг 5. Controller

@RestController
@RequestMapping("/api/v1/events")
public class EventController {
    private final EventService eventService;

    public EventController(EventService eventService) { this.eventService = eventService; }

    @PostMapping
    public ResponseEntity<EventResponse> create(@Valid @RequestBody CreateEventRequest req) {
        EventResponse created = eventService.create(req);
        URI location = URI.create("/api/v1/events/" + created.id());
        return ResponseEntity.created(location).body(created);
    }

    @GetMapping
    public List<EventResponse> findAll() { return eventService.findAll(); }

    @GetMapping("/{id}")
    public EventResponse findById(@PathVariable Long id) { return eventService.findById(id); }

    @PutMapping("/{id}")
    public EventResponse update(@PathVariable Long id,
                                @Valid @RequestBody UpdateEventRequest req) {
        return eventService.update(id, req);
    }

    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    public void delete(@PathVariable Long id) { eventService.delete(id); }
}

Шаг 7. GlobalExceptionHandler

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(EventNotFoundException.class)
    public ResponseEntity<ErrorResponse> handleNotFound(
            EventNotFoundException ex, HttpServletRequest req) {
        return buildResponse(HttpStatus.NOT_FOUND, ex.getMessage(), req);
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ErrorResponse> handleValidation(
            MethodArgumentNotValidException ex, HttpServletRequest req) {
        String message = ex.getBindingResult().getFieldErrors().stream()
                .map(e -> e.getField() + ": " + e.getDefaultMessage())
                .collect(Collectors.joining("; "));
        return buildResponse(HttpStatus.BAD_REQUEST, message, req);
    }

    @ExceptionHandler(MethodArgumentTypeMismatchException.class)
    public ResponseEntity<ErrorResponse> handleTypeMismatch(
            MethodArgumentTypeMismatchException ex, HttpServletRequest req) {
        String message = "Parameter '" + ex.getName() + "' has invalid value: '" + ex.getValue() + "'";
        return buildResponse(HttpStatus.BAD_REQUEST, message, req);
    }

    @ExceptionHandler(BookingException.class)
    public ResponseEntity<ErrorResponse> handleBusiness(
            BookingException ex, HttpServletRequest req) {
        return buildResponse(HttpStatus.BAD_REQUEST, ex.getMessage(), req);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorResponse> handleAll(
            Exception ex, HttpServletRequest req) {
        return buildResponse(HttpStatus.INTERNAL_SERVER_ERROR,
                "Unexpected error: " + ex.getMessage(), req);
    }

    private ResponseEntity<ErrorResponse> buildResponse(
            HttpStatus status, String message, HttpServletRequest req) {
        ErrorResponse error = new ErrorResponse(
                Instant.now(), status.value(),
                status.getReasonPhrase(), message, req.getRequestURI());
        return ResponseEntity.status(status).body(error);
    }
}

🐞 Частые ошибки на практике

1. Валидация не сработала — забыли @Valid
@PostMapping
public EventResponse create(@RequestBody CreateEventRequest req) { ... }   // ❌ @Valid нет
Аннотации @NotBlank, @Future на DTO не работают без @Valid. Spring не запускает валидатор. Решение: @Valid @RequestBody CreateEventRequest req.
2. Ожидание 404, а получаешь 500
throw new RuntimeException("Event not found");   // ❌ все RuntimeException → 500
Решение: создать EventNotFoundException extends BookingException и написать @ExceptionHandler(EventNotFoundException.class) → 404.
3. Ловля только конкретных исключений — забыли fallback
@ExceptionHandler(EventNotFoundException.class)   // только это — и всё
Любое непойманное исключение (NullPointerException, ошибка Jackson) → Whitelabel Error Page с огромным стектрейсом. Решение: добавить @ExceptionHandler(Exception.class) как страховку.
4. Каскадная валидация не работает
Если DTO содержит вложенный объект (Address, List<String>), валидация не заходит внутрь без @Valid на поле: 
public record CreateEventRequest(
    @Valid @NotNull Address address,   // ← без @Valid поля внутри не проверятся
    ...
) {}
5. @NotBlank на не-String поле
@NotBlank LocalDateTime date;   // ❌ не скомпилируется или упадёт в runtime
@NotBlank работает только со String. Для объектов — @NotNull, для дат — @Future/@Past, для коллекций — @NotEmpty.
6. Имя поля в сообщении об ошибке не совпадает с DTO
Клиент видит "name: must not be blank", а в DTO поле eventName — сбивает с толку. Решение: используй осмысленные имена полей в DTO, совпадающие с JSON.
7. Возврат void на DELETE без @ResponseStatus
@DeleteMapping("/{id}")
public void delete(@PathVariable Long id) { ... }   // → 200 OK, а не 204
Spring по умолчанию вернёт 200. Решение: добавить @ResponseStatus(HttpStatus.NO_CONTENT).
8. Двойная валидация — и в контроллере, и в сервисе
Валидация должна быть в одном месте. Если DTO валидируется через @Valid — не дублируй проверки в сервисе. Бизнес-валидация (например, «такое событие уже есть») — в сервисе.
9. @RestControllerAdvice не подхватился
Класс должен быть в пакете, сканируемом Spring (от @SpringBootApplication вниз). Если положить в com.other.package — не сработает. Решение: держи внутри основного пакета.
10. Отсутствие Location header на 201 Created
REST-стандарт: при POST в ответе должен быть заголовок Location с URL созданного ресурса. Без него клиент не знает, где искать. Решение: ResponseEntity.created(URI.create("/api/v1/events/" + id)).body(...).

❓ Вопросы на собеседование

1. Что такое слоистая архитектура и зачем она нужна?

Разделение кода на слои: Controller (HTTP) → Service (бизнес-логика) → Repository (данные). Каждый слой знает только о соседе снизу. Плюсы: 1) изоляция — сменить БД без переписывания сервиса, 2) тестируемость — можно замокать слой, 3) переиспользование — сервис дёргается из REST, Kafka, CLI. Слоистая архитектура — основа enterprise-приложений.

2. Чем @Valid отличается от ручной валидации?

@Valid — декларативная валидация через аннотации на DTO (@NotBlank, @Future). Spring сам запускает валидатор при поступлении запроса, до вызова метода. Плюсы: 1) меньше кода, 2) валидация рядом с полем, 3) стандартизированный ответ. Ручная валидация — гибче (можно проверить бизнес-правила), но многословнее. Обычно комбинация: формат — через Bean Validation, бизнес-правила — в сервисе.

3. Что делает @RestControllerAdvice и почему это лучше try/catch?

@RestControllerAdvice — AOP-механизм, перехватывающий исключения из всех контроллеров. Централизует обработку ошибок в одном классе. Лучше try/catch потому что: 1) нет дублирования в 100 методах, 2) единый формат ответа, 3) невозможно забыть обработать — @ExceptionHandler(Exception.class) как страховка, 4) тестируется один раз на все контроллеры.

4. Какие HTTP-статусы использовать для CRUD?

POST → 201 Created + Location header. GET → 200 OK, для несуществующего → 404 Not Found. PUT → 200 OK (с обновлённым объектом). DELETE → 204 No Content. Ошибки валидации → 400 Bad Request. Конфликты (дубликат) → 409 Conflict. Внутренние ошибки → 500 Internal Server Error.

5. Зачем иерархия исключений: BookingException → EventNotFoundException?

BookingException — базовый класс для всех бизнес-исключений. Отделяет «наши осознанные» ошибки от технических (NullPointerException, DataAccessException). Позволяет: 1) поймать всё «своё» одним @ExceptionHandler(BookingException.class), 2) для каждого наследника уточнить статус (404, 409), 3) расширять — новый тип ошибки наследуется от BookingException и попадает в общую обработку.

6. Что вернётся, если в DTO стоит @NotBlank, но клиент прислал null?

@NotBlank ловит null, пустую строку и строку из пробелов. Spring бросит MethodArgumentNotValidException. Наш @ExceptionHandler вернёт 400 Bad Request с сообщением "name: Name must not be blank". Метод контроллера не вызовется — валидация работает на входе.

7. Почему DTO Request и DTO Response — разные классы?

Потому что вход и выход API — это разные контракты. В Request нет id (его генерит сервер), нет status (устанавливается логикой). В Response нет валидации (она нужна только на входе). Разделение защищает от: 1) клиент подсовывает свой id, 2) клиент меняет поля, которые не должен. Плюс — можно менять DTO независимо: добавили поле в Response — Request не тронут.

8. Что такое ResponseEntity и когда использовать вместо обычного возврата?

ResponseEntity<T> — обёртка над ответом, даёт полный контроль: статус, headers, body. Используй, когда: 1) нужен нестандартный статус (201, 204), 2) нужен Location header, 3) статус зависит от логики. Обычный возврат (return eventResponse;) — для простых случаев, статус по умолчанию 200 OK. @ResponseStatus — для фиксированного статуса без гибкости.
